# 房源图片分析系统改造说明

## 改造目标

将现有的房源分析系统改造为支持结构化特征识别的系统，通过图片识别房间内的具体设施和特征，并将结果以JSON格式存储在数据库的`content`字段中。

## 改造范围说明

- 本次改造仅涉及“内容生成阶段”（内容结构、存储与接口返回格式）。
- “房间图片识别阶段”保持不变：
  - 现有 `app/services/gemini_service.py` 中的房间判定逻辑（如 `analyze_image_with_gemini`）不做修改。
  - 现有 `app/services/image_service.py` 的图片下载、并发控制与判定流程不做修改。
- 与识别链路的衔接方式不变，仅在识别完成后进入新的内容生成与存储流程。
- 不影响已有调用方对“是否房间”的判断与使用，改造后依旧兼容原有识别结果产出。

## 新的Content字段格式

### 数据结构
```json
{
  "阳台": false,
  "独卫": false,
  "飘窗": false,
  "开间": false,
  "loft": false,
  "马桶": false,
  "蹲便": false,
  "上下铺": false,
  "精装": false
}
```

### 字段说明
- 所有字段默认值为 `false`
- 通过图片识别后，将识别到的特征对应的字段设置为 `true`
- **精装字段**: `true` 表示精装修，`false` 表示简装修或无装修
- 系统只关注值为 `true` 的字段，表示房间具备该特征

```python
class RoomFeatures(BaseModel):
    """房间特征模型"""
    阳台: bool = False
    独卫: bool = False
    飘窗: bool = False
    开间: bool = False
    loft: bool = False
    马桶: bool = False
    蹲便: bool = False
    上下铺: bool = False
    精装: bool = False  # true为精装，false为简装或无装修
    
    def get_true_features(self) -> List[str]:
        """获取所有为true的特征"""
        return [key for key, value in self.dict().items() if value]
    
    def to_json_string(self) -> str:
        """转换为JSON字符串"""
        return self.json(ensure_ascii=False)
    
    @classmethod
    def from_json_string(cls, json_str: str) -> 'RoomFeatures':
        """从JSON字符串创建实例"""
        return cls.parse_raw(json_str)
```

```python
async def analyze_room_features(image_data: bytes, mime_type: str, image_url: str, request_id: str) -> dict:
    """分析房间特征"""
    try:
        prompt = get_room_feature_analysis_prompt()
        
        # 调用Gemini API进行分析
        response = await call_gemini_api(image_data, mime_type, prompt, request_id)
        
        # 解析JSON响应
        features = json.loads(response)
        
        # 验证字段完整性
        expected_fields = ["阳台", "独卫", "飘窗", "开间", "loft", "马桶", "蹲便", "上下铺", "精装"]
        for field in expected_fields:
            if field not in features:
                features[field] = False
        
        return features
        
    except Exception as e:
        logger.error(f"房间特征分析失败", request_id=request_id, error=str(e))
        # 返回默认值
        return {
            "阳台": False, "独卫": False, "飘窗": False, "开间": False, "loft": False,
            "马桶": False, "蹲便": False, "上下铺": False, "精装": False
        }
```

async def process_room_analysis(room_id: str, business_type: str, urls: List[str], request_id: str):
    """处理房间分析任务"""
    try:
        # 更新状态为处理中
        await update_processing_status(room_id, "processing")
        
        # 处理所有图片
        results = await process_batch_images_for_features(urls, request_id)
        
        # 合并所有图片的特征结果
        merged_features = merge_features_from_multiple_images(results)
        
        # 转换为JSON字符串
        content_json = json.dumps(merged_features, ensure_ascii=False)
        
        # 保存到数据库
        await save_analysis_result(room_id, business_type, content_json, "completed")
        
    except Exception as e:
        await save_analysis_result(room_id, business_type, None, "failed")
        raise e

    def merge_features_from_multiple_images(results: List[dict]) -> dict:
        """合并多张图片的特征结果"""
        merged = {
            "阳台": False, "独卫": False, "飘窗": False, "开间": False, "loft": False,
            "马桶": False, "蹲便": False, "上下铺": False, "精装": False
        }
    
    for result in results:
        if result['success'] and 'features' in result:
            features = result['features']
            for key, value in features.items():
                if key in merged and value:
                    merged[key] = True
    
    return merged
    
    async def get_existing_analysis(self, room_id: str) -> Optional[RoomAnalysisHistory]:
        """获取现有分析记录"""
        try:
            result = await get_analysis_result(room_id)
            if result and result['content']:
                # 解析现有的分析历史
                content_data = json.loads(result['content'])
                if 'image_records' in content_data:
                    # 返回完整的历史记录
                    history = RoomAnalysisHistory(**content_data)
                    # 检查是否已删除
                    if history.is_delete == 1:
                        logger.info(f"房间分析记录已删除: {room_id}")
                        return None
                    return history
                else:
                    # 兼容旧格式，转换为新格式
                    features = RoomFeatures(**content_data)
                    return RoomAnalysisHistory(
                        room_id=room_id,
                        business_type=result['business_type'],
                        current_features=features,
                        image_records=[],
                        last_updated=result['updated_at'],
                        version=1,
                        is_delete=0
                    )
            return None
        except Exception as e:
            logger.error(f"获取现有分析记录失败: {room_id}, 错误: {e}")
            return None
    
    async def save_incremental_update(self, room_id: str, business_type: str, 
                                    analysis_history: RoomAnalysisHistory) -> None:
        """保存增量更新结果"""
        try:
            # 转换为JSON字符串
            content_json = analysis_history.json(ensure_ascii=False)
            
            # 更新数据库
            await update_room_analysis_status(room_id, "completed", content_json)
            
            logger.info(f"增量更新保存成功: {room_id}")
            
        except Exception as e:
            logger.error(f"保存增量更新失败: {room_id}, 错误: {e}")
            raise e

#### 统一分析API端点
```python
@router.post("/analyze_room")
async def analyze_rooms(request: AnalyzeRoomRequest):
    """分析房间图片（支持单个和批量）"""
    try:
        logger.info(f"收到批量分析请求: {len(request.rooms)} 个房间")
        
        results = []
        submitted_count = 0
        failed_count = 0
        
        # 并行处理所有房间
        tasks = []
        for room_item in request.rooms:
            task = process_single_room_analysis(
                room_item.roomId, 
                room_item.businessType, 
                room_item.imageUrl
            )
            tasks.append(task)
        
        # 等待所有任务完成
        room_results = await asyncio.gather(*tasks, return_exceptions=True)
        
        # 处理结果
        for i, result in enumerate(room_results):
            room_item = request.rooms[i]
            
            if isinstance(result, Exception):
                failed_count += 1
                results.append({
                    "room_id": room_item.roomId,
                    "business_type": room_item.businessType,
                    "success": False,
                    "error": str(result)
                })
            else:
                submitted_count += 1
                results.append({
                    "room_id": room_item.roomId,
                    "business_type": room_item.businessType,
                    "success": True,
                    "processing_status": "submitted"
                })
        
        return AnalyzeRoomResponse(
            success=True,
            total_rooms=len(request.rooms),
            submitted_rooms=submitted_count,
            failed_rooms=failed_count,
            results=results
        )
        
    except Exception as e:
        logger.error(f"房间分析失败: {e}")
        return AnalyzeRoomResponse(
            success=False,
            total_rooms=len(request.rooms),
            submitted_rooms=0,
            failed_rooms=len(request.rooms),
            results=[],
            error=str(e)
        )

async def process_single_room_analysis(room_id: str, business_type: str, image_url: str):
    """处理单个房间分析"""
    try:
        # 验证业务类型
        if business_type not in ["whole_rent", "centralized", "shared_rent"]:
            raise ValueError(f"不支持的业务类型: {business_type}")
        
        # 创建分析任务
        await async_processor.process_features_with_optimization(
            room_id=room_id,
            business_type=business_type,
            image_urls=[image_url],
            request_id=f"batch_{room_id}_{int(time.time())}"
        )
        
        return {"status": "submitted"}
        
    except Exception as e:
        logger.error(f"房间 {room_id} 分析失败: {e}")
        raise e
```

```python
async def get_room_analysis_status(room_ids: List[str]) -> List[RoomAnalysisResult]:
    """获取房间分析状态（内部函数）"""
    results = []
    
    # 并行查询所有房间状态

        tasks.append(task)
```

```python
async def save_analysis_result(room_id: str, business_type: str, content: str, status: str):
    """保存分析结果到数据库"""
    async with get_db_connection() as conn:
        await conn.execute("""
            INSERT INTO qft_ai_room_analysis (room_id, business_type, content, processing_status)
            VALUES (%s, %s, %s, %s)
            ON DUPLICATE KEY UPDATE
            content = VALUES(content),
            processing_status = VALUES(processing_status),
            updated_at = CURRENT_TIMESTAMP
        """, (room_id, business_type, content, status))
```

### 查询分析结果

```python
async def get_analysis_result(room_id: str):
    """查询分析结果"""
    async with get_db_connection() as conn:
        result = await conn.fetchrow("""
            SELECT room_id, business_type, content, processing_status, created_at, updated_at
            FROM qft_ai_room_analysis
            WHERE room_id = %s AND is_delete = 0
        """, (room_id,))
        return result

async def soft_delete_analysis_result(room_id: str):
    """软删除分析结果"""
    async with get_db_connection() as conn:
        await conn.execute("""
            UPDATE qft_ai_room_analysis 
            SET is_delete = 1, updated_at = CURRENT_TIMESTAMP
            WHERE room_id = %s
        """, (room_id,))
        logger.info(f"软删除分析结果: {room_id}")

async def soft_delete_image_from_analysis(room_id: str, image_url: str):
    """软删除指定图片的分析记录"""
    try:
        # 获取现有分析记录
        existing_analysis = await get_existing_analysis(room_id)
        if existing_analysis:
            # 软删除指定图片
            if existing_analysis.soft_delete_image_record(image_url):
                # 保存更新后的记录
                content_json = existing_analysis.json(ensure_ascii=False)
                await update_room_analysis_status(room_id, "completed", content_json)
                logger.info(f"软删除图片记录: {room_id}, {image_url}")
                return True
        return False
    except Exception as e:
        logger.error(f"软删除图片记录失败: {room_id}, {image_url}, 错误: {e}")
        return False
```

```python
class EnhancedAsyncFeatureProcessor(AsyncContentProcessor):
    """增强的特征处理器"""
    
    async def process_features_with_optimization(self, room_id: str, business_type: str, 
                                               image_urls: List[str], request_id: str, 
                                               incremental: bool = True) -> None:
        """优化的特征处理流程（支持增量处理）"""
        try:
            # 1. 检查现有分析记录
            existing_analysis = await self.get_existing_analysis(room_id)
            
            if incremental and existing_analysis:
                # 增量处理模式
                await self.process_incremental_update(room_id, business_type, image_urls, 
                                                    existing_analysis, request_id)
            else:
                # 全量处理模式
                await self.process_full_analysis(room_id, business_type, image_urls, request_id)
            
        except Exception as e:
            await self.handle_processing_error(room_id, e)
    
    async def process_incremental_update(self, room_id: str, business_type: str, 
                                       image_urls: List[str], existing_analysis: RoomAnalysisHistory, 
                                       request_id: str) -> None:
        """增量更新处理"""
        try:
            logger.info(f"开始增量更新: {room_id}, 总图片数: {len(image_urls)}")
            
            # 1. 识别新图片
            new_images = existing_analysis.get_new_images(image_urls)
            
            if not new_images:
                logger.info(f"没有新图片需要处理: {room_id}")
                return
            
            logger.info(f"发现 {len(new_images)} 张新图片需要分析: {room_id}")
            
            # 2. 只处理新图片
            new_feature_results = await self.process_images_parallel(new_images, request_id)
            
            # 3. 合并新特征与现有特征
            merged_features = self.merge_incremental_features(new_feature_results, existing_analysis.current_features)
            
            # 4. 更新分析历史
            for i, result in enumerate(new_feature_results):
                if result['success'] and 'features' in result:
                    image_url = new_images[i]
                    features = RoomFeatures(**result['features'])
                    existing_analysis.add_image_analysis(image_url, features)
            
            # 5. 保存更新后的结果
            await self.save_incremental_update(room_id, business_type, existing_analysis)
            
            logger.info(f"增量更新完成: {room_id}, 处理新图片: {len(new_images)}")
            
        except Exception as e:
            logger.error(f"增量更新失败: {room_id}, 错误: {e}")
            raise e
    
    async def process_full_analysis(self, room_id: str, business_type: str, 
                                  image_urls: List[str], request_id: str) -> None:
        """全量分析处理"""
        try:
            logger.info(f"开始全量分析: {room_id}")
            
            # 1. 创建初始记录
            await self.create_initial_record(room_id, business_type)
            
            # 2. 并行处理所有图片
            feature_results = await self.process_images_parallel(image_urls, request_id)
            
            # 3. 智能合并特征结果
            merged_features = self.merge_features_intelligently(feature_results)
            
            # 4. 验证和优化结果
            validated_features = self.validate_and_optimize_features(merged_features)
            
            # 5. 保存到数据库
            await self.save_features_to_database(room_id, business_type, validated_features)
            
        except Exception as e:
            logger.error(f"全量分析失败: {room_id}, 错误: {e}")
            raise e
    
    def merge_incremental_features(self, new_results: List[dict], existing_features: RoomFeatures) -> RoomFeatures:
        """合并增量特征"""
        merged_features = existing_features.copy()
        
        for result in new_results:
            if result['success'] and 'features' in result:
                new_features = RoomFeatures(**result['features'])
                merged_features = new_features.merge_with_existing_features(merged_features)
        
        return merged_features
    
    async def process_images_parallel(self, image_urls: List[str], request_id: str) -> List[dict]:
        """并行处理多张图片"""
        tasks = [self.process_single_image_features(url, request_id) for url in image_urls]
        results = await asyncio.gather(*tasks, return_exceptions=True)
        
        # 处理异常结果
        processed_results = []
        for i, result in enumerate(results):
            if isinstance(result, Exception):
                logger.error(f"图片{i+1}处理失败", request_id=request_id, error=str(result))
                processed_results.append(get_default_features())
            else:
                processed_results.append(result)
        
        return processed_results
    
    def merge_features_intelligently(self, feature_results: List[dict]) -> dict:
        """智能合并特征结果"""
        merged = get_default_features()
        confidence_threshold = 0.7  # 置信度阈值
        
        for result in feature_results:
            if result['success'] and 'features' in result:
                features = result['features']
                confidence_scores = features.get('confidence_scores', {})
                
                for feature_name, is_detected in features.items():
                    if feature_name in ['confidence_scores', 'feature_categories']:
                        continue
                    
                    confidence = confidence_scores.get(feature_name, 0.5)
                    
                    # 如果置信度高于阈值且检测到特征，则合并
                    if confidence >= confidence_threshold and is_detected:
                        merged[feature_name] = True
                        # 更新置信度（取最高值）
                        if 'confidence_scores' not in merged:
                            merged['confidence_scores'] = {}
                        merged['confidence_scores'][feature_name] = max(
                            merged['confidence_scores'].get(feature_name, 0),
                            confidence
                        )
        
        return merged
```

```python
class FeatureAnalysisMonitor:
    """特征分析监控器"""
    
    def __init__(self):
        self.analysis_stats = {
            'total_requests': 0,
            'successful_requests': 0,
            'failed_requests': 0,
            'feature_detection_rates': {},
            'average_processing_time': 0,
            'error_types': {}
        }
    
    async def record_analysis_result(self, room_id: str, features: dict, 
                                   processing_time: float, success: bool, error_type: str = None):
        """记录分析结果"""
        self.analysis_stats['total_requests'] += 1
        
        if success:
            self.analysis_stats['successful_requests'] += 1
            # 记录特征检测率
            for feature, is_detected in features.items():
                if feature not in ['confidence_scores', 'feature_categories']:
                    if feature not in self.analysis_stats['feature_detection_rates']:
                        self.analysis_stats['feature_detection_rates'][feature] = {'detected': 0, 'total': 0}
                    self.analysis_stats['feature_detection_rates'][feature]['total'] += 1
                    if is_detected:
                        self.analysis_stats['feature_detection_rates'][feature]['detected'] += 1
        else:
            self.analysis_stats['failed_requests'] += 1
            if error_type:
                self.analysis_stats['error_types'][error_type] = self.analysis_stats['error_types'].get(error_type, 0) + 1
        
        # 更新平均处理时间
        current_avg = self.analysis_stats['average_processing_time']
        total_requests = self.analysis_stats['total_requests']
        self.analysis_stats['average_processing_time'] = (current_avg * (total_requests - 1) + processing_time) / total_requests
    
    def get_analysis_statistics(self) -> dict:
        """获取分析统计信息"""
        stats = self.analysis_stats.copy()
        
        # 计算特征检测率
        for feature, data in stats['feature_detection_rates'].items():
            if data['total'] > 0:
                data['detection_rate'] = data['detected'] / data['total']
        
        # 计算成功率
        if stats['total_requests'] > 0:
            stats['success_rate'] = stats['successful_requests'] / stats['total_requests']
        
        return stats
```
